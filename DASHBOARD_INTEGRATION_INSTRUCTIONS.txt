# STEP 5: Build Paper Trading Display
# Add this method to the UnifiedDashboard class:
# ============================================

def _build_paper_trading_display(self, data: Dict) -> html.Div:
    """Build paper trading performance display."""
    
    # Get paper trading stats
    if not self.paper_trading:
        return create_empty_state("Paper trading not started")
    
    try:
        stats = self.paper_trading.get_all_stats()
    except:
        return create_empty_state("Loading paper trading data...")
    
    overall = stats.get('overall', {})
    
    # Three scenario cards
    scenario_cards = []
    for name in ['aggressive', 'moderate', 'conservative']:
        if name in stats:
            s = stats[name]
            
            # Color based on P&L
            pnl_color = '#27ae60' if s['pnl'] > 0 else '#e74c3c' if s['pnl'] < 0 else '#95a5a6'
            
            card = html.Div([
                html.H4(name.upper(), style={'color': '#2c3e50', 'marginBottom': '10px'}),
                
                html.Div([
                    html.Div([
                        html.Strong("Equity: "),
                        html.Span(f"${s['equity']:,.2f}", style={'color': pnl_color, 'fontSize': '1.2em'})
                    ], style={'marginBottom': '5px'}),
                    
                    html.Div([
                        html.Strong("P&L: "),
                        html.Span(f"${s['pnl']:+,.2f} ({s['pnl_percent']:+.2f}%)", 
                                 style={'color': pnl_color})
                    ], style={'marginBottom': '5px'}),
                    
                    html.Div([
                        html.Strong("Trades: "),
                        html.Span(f"{s['total_trades']} ({s['wins']}W / {s['losses']}L)")
                    ], style={'marginBottom': '5px'}),
                    
                    html.Div([
                        html.Strong("Win Rate: "),
                        html.Span(f"{s['win_rate']:.1f}%")
                    ], style={'marginBottom': '5px'}),
                    
                    html.Hr(),
                    
                    html.Div([
                        html.Strong("Position: "),
                        html.Span("✅ IN TRADE" if s['in_position'] else "⚪ FLAT", 
                                 style={'color': '#27ae60' if s['in_position'] else '#95a5a6'})
                    ])
                ])
                
            ], style={
                'padding': '15px',
                'backgroundColor': '#ecf0f1',
                'borderRadius': '8px',
                'border': '1px solid #bdc3c7'
            })
            
            scenario_cards.append(card)
    
    # Overall performance
    overall_pnl_color = '#27ae60' if overall.get('total_pnl', 0) > 0 else '#e74c3c'
    
    return html.Div([
        # Three scenario cards
        html.Div(scenario_cards, 
                style={'display': 'grid', 'gridTemplateColumns': '1fr 1fr 1fr', 
                      'gap': '15px', 'marginBottom': '15px'}),
        
        # Overall stats
        html.Div([
            html.H4("OVERALL PERFORMANCE", style={'color': '#2c3e50', 'marginBottom': '10px'}),
            html.Div([
                html.Span(f"Total Equity: ${overall.get('total_equity', 0):,.2f}", 
                         style={'marginRight': '20px', 'fontSize': '1.1em'}),
                html.Span(f"Total P&L: ", style={'marginRight': '5px'}),
                html.Span(f"${overall.get('total_pnl', 0):+,.2f} ({overall.get('total_pnl_percent', 0):+.2f}%)", 
                         style={'color': overall_pnl_color, 'fontSize': '1.2em', 'fontWeight': 'bold'})
            ])
        ], style={
            'padding': '15px',
            'backgroundColor': '#ecf0f1',
            'borderRadius': '8px',
            'textAlign': 'center'
        })
    ])
